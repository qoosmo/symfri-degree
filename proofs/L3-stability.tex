\section{Folding preserves the graded codes}

\setcounter{lemma}{2}\begin{lemma}[Stability]\label{lem:stab}
Let $1\le i\le m$, $w\in\N^m$, $D\in\N$, $r\in\F$ and $c:=rT^{w_i}$.
\begin{enumerate}
\item[(a)] If $g\in\F[T][x_i,\dots,x_m]_{<k}$ and $\gdeg_w g\le D$, then
$g(c,x_{i+1},\dots,x_m)\in\F[T][x_{i+1},\dots,x_m]_{<k}$ and its graded degree is $\le D$.
\item[(b)] $\Fold_c\big(\Vc_{i-1}^{w,D}\big)\subseteq\Vc_i^{w,D}$.
\item[(c)] For $c_l:=r_lT^{w_l}$ ($1\le l\le i$),
$\Fold_{c_i}\circ\cdots\circ\Fold_{c_1}\big(C_{w,D}\big)$ consists of words of $\Vc_i^{w,D}$ whose
values have $T$-degree at most
\[
d_i:=\min\Big(D,\ (k-1)\sum_{l\le i}w_l\Big).
\]
\end{enumerate}
\end{lemma}

\begin{proof}
(a) The substitution maps the monomial $T^a x_i^{\,j}x^\beta$ (with $\beta$ on
$x_{i+1},\dots,x_m$) to $r^j\,T^{a+w_ij}x^\beta$. The exponents of $x_{i+1},\dots,x_m$ are
unchanged, hence $<k$, and the graded degree $a+w_ij+\sum_{l>i}w_l\beta_l$ is unchanged,
hence $\le D$. A sum of such terms has graded degree $\le D$.

(b) Combine (a) with \cref{lem:fold-sub}(b).

(c) Start from a $T$-free $f$ with $\deg_w f\le D$. By \cref{lem:fold-sub}(b) applied
$i$ times, the result is $U^{(i)}_{g}$ with $g:=f(c_1,\dots,c_i,x_{i+1},\dots,x_m)$, which lies
in $\Vc_i^{w,D}$ by (a). A monomial $x^\alpha$ of $f$ contributes the $T$-exponent
$\sum_{l\le i}w_l\alpha_l\le(k-1)\sum_{l\le i}w_l$, and this exponent is at most the graded
degree, hence $\le D$.
\end{proof}
