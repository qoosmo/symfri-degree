\section{The subcode lemma}

\setcounter{lemma}{6}\begin{lemma}[Subcode lemma]\label{lem:subcode}
Let $\E\supseteq\F$, let $V$ be an $\E$-linear code and $W\subseteq V$ an
$\F$-linear subspace. Let $v_0,\dots,v_l\in V$ and $S'\subseteq\F$ with $|S'|>l$. If
\[
\textstyle\sum_{j\le l}z^jv_j\in W\qquad\text{for every }z\in S',
\]
then $v_j\in W$ for every $j$.
\end{lemma}

\begin{proof}
Pick distinct $z_0,\dots,z_l\in S'$ and put $P_a:=\sum_jz_a^jv_j\in W$. The Vandermonde
matrix $A:=(z_a^{\,j})_{a,j}\in\F^{(l+1)\times(l+1)}$ is invertible, so
$v_j=\sum_a(A^{-1})_{j,a}P_a$ with coefficients $(A^{-1})_{j,a}\in\F$. Since $W$ is
$\F$-linear, $v_j\in W$.
\end{proof}
\uses{nothing}\usedby{\cref{cor:subcode}}

\begin{corollary}[Correlated agreement for linear subcodes]\label{cor:subcode}
In the setting of \cref{lem:subfield}, let $W\subseteq V$ be an $\F$-linear subspace, and
assume that for every $z\in S$ some $P_z\in W$ satisfies
$\agree_\mu(\sum_jz^ju_j,P_z)\ge\alpha$. Then there are $v_0,\dots,v_l\in W$ with
$\mu\big(\{x:\ u_j(x)=v_j(x)\ \forall j\}\big)\ge\alpha$.
\end{corollary}

\begin{proof}
Apply \cref{lem:subfield} with these $P_z\in W$: it gives $v_j\in V$ with (1), and
$\sum_jz^jv_j=P_z\in W$ for all $z\in S'$, $|S'|>l$. Then $v_j\in W$ by \cref{lem:subcode}.
\end{proof}
\uses{\cref{lem:subfield,lem:subcode}}\usedby{L8}

\begin{remark}
\Cref{cor:subcode} says that weighted correlated agreement over curves is inherited by
every $\F$-linear subcode, with the same parameters, as long as the curve parameter is
drawn from $\F$. Nothing in it is specific to degree constraints; it applies equally to
subcodes defined by linear constraints such as the constrained Reed--Solomon codes of
WHIR.
\end{remark}
