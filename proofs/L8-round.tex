\section{The round lemma}

Throughout, $\kappa\ge3$, $\rho:=k^m/N$ and $\alpha_0:=\sqrt\rho\,(1+\frac1{2\kappa})$.

\setcounter{lemma}{7}\begin{lemma}[Round lemma]\label{lem:round}
Fix $1\le i\le m$, a word $u:L_{i-1}\to\F[T]_{\le d_{i-1}}$ and a weight
$\mu^{(i-1)}$ on $L_{i-1}$ with values in $\frac1{k^{i-1}}\mathbb Z$; let $\nu^{(i)}$ be its
fibre average (\cref{def:weights}). Put
\[
\alpha_i:=\max\Big(\agree_{\mu^{(i-1)}}\big(u,\Vc_{i-1}^{w,D}\big),\ \alpha_0\Big),\qquad
S:=\Big\{r\in\F:\ \agree_{\nu^{(i)}}\big(\Fold_{rT^{w_i}}(u),\Vc_i^{w,D}\big)>\alpha_i\Big\}.
\]
Then
\[
|S|\le(k-1)\max\Big(\frac{(1+\frac1{2\kappa})^7\kappa^7}{3\rho^{3/2}}\,|L_i|^2,\ \
\frac{2\kappa+1}{\sqrt\rho}\,(N+1)\Big).
\]
\end{lemma}

\begin{proof}
Assume the contrary.

\emph{Step 1 (the fold is a curve).} With $u_j$ the components of $u$ and
$W_j:=T^{w_ij}u_j$, \cref{def:fold} gives $\Fold_{rT^{w_i}}(u)=\sum_{j<k}r^jW_j$. Every
$W_j$ takes values in $\F[T]_{<t}$, $t:=1+\max\big(D,\,d_{i-1}+w_i(k-1)\big)$, and so does
every word of $\Vc_i^{w,D}$.

\emph{Step 2 (correlated agreement inside the graded code).} Agreements here are
multiples of $1/N$; let $\alpha'$ be the least multiple of $1/N$ with $\alpha'>\alpha_i$, so
that $S$ is the set where the agreement is $\ge\alpha'$, and $\alpha'\ge\alpha_0$. For each
$r\in S$ fix $P_r\in\Vc_i^{w,D}$ with $\agree_{\nu^{(i)}}(\sum_jr^jW_j,P_r)\ge\alpha'$.
Transport everything by the map $\varphi$ of \cref{lem:ext} (degree-$t$ extension $\E$):
$V:=\varphi\big(\RS[\F,L_i,k^{m-i}]^t\big)=\RS[\E,L_i,k^{m-i}]$ and
$W:=\varphi(\Vc_i^{w,D})$, an $\F$-linear subspace of $V$; $\varphi$ preserves agreements and
$\F$-linear combinations. Apply \cref{cor:subcode} with $l=k-1$, $n=|L_i|$,
$M=k^i$ (so $Mn=N$): there are $v_0,\dots,v_{k-1}\in\Vc_i^{w,D}$ and a set
$A:=\{y\in L_i:\ W_j(y)=v_j(y)\ \forall j\}$ with $\nu^{(i)}(A)\ge\alpha'$.

\emph{Step 3 (divisibility).} Since $\nu^{(i)}\le1$,
$|A|\ge\nu^{(i)}(A)\,|L_i|\ge\alpha'|L_i|>\rho|L_i|=k^{m-i}$. For $a<w_ij$, the
$T^a$-coefficient of $v_j$ is a codeword of $\RS[\F,L_i,k^{m-i}]$ that vanishes on $A$
(because $W_j$ is divisible by $T^{w_ij}$), hence it is $0$. So $v_j=T^{w_ij}p_j$
pointwise, and on $A$ we get $u_j=p_j$ (cancel $T^{w_ij}$ in the domain $\F[T]$).

\emph{Step 4 (unfold).} By \cref{lem:unfold}, $h(x):=\sum_jx^jp_j(x^k)$ lies in
$\Vc_{i-1}^{w,D}$. For $y\in A$ and $x\in\fib(y)$,
\[
u(x)=\sum_{j<k}x^ju_j(y)=\sum_{j<k}x^jp_j(y)=h(x),
\]
where the first equality is the inversion of \cref{def:fold}:
$\sum_jx^ju_j(y)=\frac1k\sum_{x'\in\fib(y)}u(x')\sum_{j<k}(x/x')^j=u(x)$, since
$x/x'\in\mu_k$.

\emph{Step 5 (contradiction).} Hence
\[
\agree_{\mu^{(i-1)}}(u,h)\ \ge\ \frac1{|L_{i-1}|}\sum_{y\in A}\sum_{x\in\fib(y)}\mu^{(i-1)}(x)
=\frac1{|L_i|}\sum_{y\in A}\nu^{(i)}(y)=\nu^{(i)}(A)\ \ge\ \alpha'\ >\ \alpha_i ,
\]
contradicting $\agree_{\mu^{(i-1)}}(u,\Vc_{i-1}^{w,D})\le\alpha_i$.
\end{proof}
\uses{\cref{def:fold,def:weights,lem:ext,cor:subcode,lem:unfold}}\usedby{\cref{thm:main}}

\begin{remark}[Unique-decoding version]\label{rem:round-ud}
Put $\alpha^{\mathrm U}_0:=\frac{1+\rho}2$ and
$\eta_{\mathrm U}:=\min\big(\frac{(1-\sqrt\rho)^2}2,\frac{\sqrt\rho}{20}\big)$. If
$\alpha_0$ is replaced by $\alpha^{\mathrm U}_0$ in the definition of $\alpha_i$, then
\[
|S|\le(k-1)\,(N+1)\Big(\frac1{\eta_{\mathrm U}}+\frac3{\sqrt\rho}\Big).
\]
Same proof, using \cref{rem:subfield71}: here $\alpha'>\alpha^{\mathrm U}_0$, so
$\eta\ge\eta_{\mathrm U}$, and $(\star\star)$ reads $|S|>(k-1)|L_i|$ and
$|S|\ge(k-1)(N+1)(\frac1\eta+\frac3{\sqrt\rho})$; the second condition implies the first.
\end{remark}
