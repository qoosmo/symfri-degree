\section{Fold equals substitution}

\begin{lemma}[Components of a polynomial]\label{lem:fold-parts}
Let $1\le i\le m$, let $R$ be a commutative $\F$-algebra, and let
$P(Y)=\sum_{j<k}Y^jP_j(Y^k)$ with $P_j\in R[Y]$. Then the components of $u:=P|_{L_{i-1}}$
(\cref{def:fold}) are $u_j=P_j|_{L_i}$ for $0\le j<k$.
\end{lemma}

\begin{proof}
Fix $y\in L_i$ and $x_0\in\fib(y)$, so $\fib(y)=x_0\Omega_k$; then
\[
\sum_{x\in\fib(y)}x^{-j}P(x)=\sum_{j'<k}P_{j'}(y)\,x_0^{\,j'-j}\sum_{\zeta\in\Omega_k}
\zeta^{\,j'-j}=k\,P_j(y),
\]
the inner sum being $k$ for $j'=j$ and $0$ otherwise by \cref{lem:rootsum}, and $k$ is invertible by
\cref{def:domains}.
\end{proof}
\uses{\cref{def:fold,def:domains,lem:rootsum}}\usedby{\cref{lem:fold-sub,lem:unfold}}

\begin{lemma}[Fold equals substitution]\label{lem:fold-sub}
Let $1\le i\le m$, $q\in R[x_i,\dots,x_m]_{<k}$ and $u:=U^{(i-1)}_q|_{L_{i-1}}$. Then for every
$c\in R$,
\[
\Fold_c(u)=U^{(i)}_{q(c,\,x_{i+1},\dots,x_m)}\big|_{L_i}.
\]
\end{lemma}

\begin{proof}
Write $q=\sum_{j<k}x_i^jq_j$; by \cref{def:kron} $U^{(i-1)}_q(Y)=\sum_jY^jU^{(i)}_{q_j}(Y^k)$, so
$u_j=U^{(i)}_{q_j}|_{L_i}$ by \cref{lem:fold-parts}, and
$\Fold_c(u)=\sum_jc^jU^{(i)}_{q_j}|_{L_i}=U^{(i)}_{\sum_jc^jq_j}|_{L_i}$.
\end{proof}
\uses{\cref{lem:fold-parts,def:kron,lem:kron}}\usedby{\cref{cor:terminal,lem:stab,lem:stab-cap}}

\begin{lemma}[Inversion]\label{lem:fold-invert}
Let $1\le i\le m$ and let $u:L_{i-1}\to R$ have components $u_0,\dots,u_{k-1}$
(\cref{def:fold}). Then $u(x)=\sum_{j<k}x^j\,u_j(x^k)$ for every $x\in L_{i-1}$.
\end{lemma}

\begin{proof}
Put $y:=x^k$; then $\sum_{j<k}x^ju_j(y)=\frac1k\sum_{x'\in\fib(y)}u(x')\sum_{j<k}(x/x')^j=u(x)$,
since $x/x'\in\Omega_k$ and the inner sum is $k$ for $x'=x$ and $0$ otherwise
(\cref{lem:rootsum}).
\end{proof}
\uses{\cref{def:fold,def:domains,lem:rootsum}}\usedby{\cref{lem:round-up}}

\begin{corollary}[Honest terminal]\label{cor:terminal}
Let $R=\F[T]$, $f\in\F[x_1,\dots,x_m]_{<k}$, $w\in\N^m$, $r\in\F^m$ and
$c_i:=r_iT^{w_i}$. Then $\Fold_{c_m}\circ\cdots\circ\Fold_{c_1}\big(U_f|_{L_0}\big)$ is the
constant $g(T):=f\big(r_1T^{w_1},\dots,r_mT^{w_m}\big)$.
\end{corollary}

\begin{proof}
Induction on $i$ with \cref{lem:fold-sub}; at $i=m$, $U^{(m)}$ is the identity.
\end{proof}
\uses{\cref{lem:fold-sub}}\usedby{\cref{prop:complete}}
