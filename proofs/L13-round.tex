\section{The round lemma}

\begin{lemma}[Descent]\label{lem:round-up}
Let $1\le i\le m$, $w\in\N^m$, $D\in\N$, let $u:L_{i-1}\to\F[T]$ have components
$u_0,\dots,u_{k-1}$ (\cref{def:fold}), let $\mu$ be a weight on $L_{i-1}$ with fibre average $\nu$
(\cref{def:fibre-avg}), and let $A\subseteq L_i$ with $\nu(A)>\rho$. If there are
$v_0,\dots,v_{k-1}\in\Gc_i^{w,D}$ with $v_j(y)=T^{w_ij}u_j(y)$ for every $y\in A$ and $j<k$, then
there is $h\in\Gc_{i-1}^{w,D}$ with $\agree_\mu(u,h)\ge\nu(A)$.
\end{lemma}

\begin{proof}\begin{enumerate}
\item Since $\nu(A)>\rho$ and $\nu\le1$ give $|A|\ge\nu(A)|L_i|>\rho|L_i|=k^{m-i}$
(\cref{def:domains}), \cref{lem:div} with $\gamma=w_ij$, $v=v_j$ and this $A$ gives $v_j=T^{w_ij}p_j$
with $p_j\in\Gc_i^{w,D-w_ij}$; cancelling in $\F[T]$ gives $u_j=p_j$ on $A$.
\item By \cref{lem:unfold}, $h:=\sum_{j<k}x^jp_j(x^k)$ lies in $\Gc_{i-1}^{w,D}$ with components
$p_j$.
\item By \cref{lem:fold-invert}, $u(x)=\sum_jx^ju_j(x^k)=\sum_jx^jp_j(x^k)=h(x)$ for every
$x\in\fib(y)$, $y\in A$.
\item By \cref{lem:lift}, $\agree_\mu(u,h)\ge\nu(A)$.
\end{enumerate}\end{proof}
\uses{\cref{def:domains,def:fibre-avg,lem:div,lem:fold-invert,lem:lift,lem:unfold}}\usedby{\cref{lem:round}}

\begin{lemma}[Round lemma]\label{lem:round}
Assume $\mathrm H(\alpha_*,\Theta)$ and $\alpha_*>\rho$. Let $1\le i\le m$, let $u:L_{i-1}\to\F[T]_{\le d_{i-1}}$,
and let $\mu^{(i-1)}$ be a weight on $L_{i-1}$ with values in $\frac1{k^{i-1}}\Z$; let
$\nu^{(i)}$ be its fibre average (\cref{def:fibre-avg}). Put
\[
\beta_i:=\max\Big(\agree_{\mu^{(i-1)}}\big(u,\Gc_{i-1}^{w,D}\big),\ \alpha_*\Big),\qquad
S_i:=\Big\{r\in\F:\ \agree_{\nu^{(i)}}\big(\Fold_{rT^{w_i}}(u),\Gc_i^{w,D}\big)>\beta_i\Big\}.
\]
Then $|S_i|\le\Theta\big(|L_i|,k^i,k-1\big)$.
\end{lemma}

\begin{proof}\begin{enumerate}
\item Assume $|S_i|>\Theta(|L_i|,k^i,k-1)$, let $u_j$ be the components of $u$ and put $\tilde u_j:=T^{w_ij}u_j$; by \cref{def:fold}, $\Fold_{rT^{w_i}}(u)=\sum_{j<k}r^j\tilde u_j$ for every $r\in\F$.
\item By \cref{lem:gran}, $S_i=\{r:\agree_{\nu^{(i)}}(\sum_jr^j\tilde u_j,\Gc_i^{w,D})\ge\beta_i^+\}$, and $\beta_i^+\ge\alpha_*$.
\item By \cref{lem:ext} with $[\E:\F]:=1+\max(D,d_{i-1}+w_i(k-1))$, the $\tilde u_j$ become words over $\E$ and $\Gc_i^{w,D}$ an $\F$-linear subspace $W$ of $\RS[\E,L_i,k^{m-i}]$, of rate $\rho$; by \cref{lem:ext-agree} the agreement of $\varphi(\tilde u_j)$ with $W$ is that of $\tilde u_j$ with $\Gc_i^{w,D}$.
\item So \cref{cor:share} with $n=|L_i|$, $M=k^i$, $l=k-1$ gives $v_0,\dots,v_{k-1}\in\Gc_i^{w,D}$ and $A\subseteq L_i$ with $\nu^{(i)}(A)\ge\beta_i^+$ and $\tilde u_j=v_j$ on $A$.
\item Since $\beta_i^+\ge\alpha_*>\rho$, \cref{lem:round-up} with $\mu:=\mu^{(i-1)}$, $\nu:=\nu^{(i)}$ and this $A$ gives $h\in\Gc_{i-1}^{w,D}$ with $\agree_{\mu^{(i-1)}}(u,h)\ge\nu^{(i)}(A)\ge\beta_i^+>\beta_i$, contradicting the definition of $\beta_i$ as a maximum over $\Gc_{i-1}^{w,D}$.
\end{enumerate}\end{proof}
\uses{\cref{hyp:ca,cor:share,lem:round-up},\cref{lem:ext,lem:ext-agree,lem:gran}}\usedby{\cref{lem:sound-fold,thm:reduction}}
