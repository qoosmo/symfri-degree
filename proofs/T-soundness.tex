\section{Soundness}

\begin{theorem}[Soundness up to the Johnson radius]\label{thm:main}
Let $\kappa\ge3$, $\rho:=k^m/N$, $\alpha\ge\alpha_0:=\sqrt\rho\,(1+\frac1{2\kappa})$, and
suppose $\agree(u_0,C_{w,D})\le\alpha$. Then for every prover, $\Pi_{w,D}$ accepts with
probability at most $\varepsilon_C+\alpha^s$, where
\[
\varepsilon_C:=\frac1{|\F|}\Big(\frac{(1+\frac1{2\kappa})^7\kappa^7\,N^2}{3(k+1)\rho^{3/2}}
+m(k-1)\frac{(2\kappa+1)(N+1)}{\sqrt\rho}\Big).
\]
\end{theorem}

\begin{proof}
\emph{Step 1 (start).} $\agree(u_0,\Vc_0^{w,D})=\agree(u_0,C_{w,D})$: if $u_0$ agrees with
$U_g|_{L_0}\in\Vc_0^{w,D}$ at $x$, then, $u_0$ being $T$-free, it agrees there with the
$T^0$-part $U_{g_0}|_{L_0}$, and $U_{g_0}|_{L_0}\in C_{w,D}$.

\emph{Step 2 (bad events).} Let $B_i$ be the event $r_i\in S_i$, with $S_i$ the set of
\cref{lem:round} for $u:=u_{i-1}$ and $\mu^{(i-1)}$; both are fixed before $r_i$ is
drawn. By \cref{lem:round},
$\sum_i\Pr[B_i]\le\varepsilon_C$, using $\sum_{i\ge1}(k-1)k^{-2i}\le\frac1{k+1}$ and
$|L_i|=N/k^i$.

\emph{Step 3 (no increase).} For every codeword $v$ on $L_i$,
$\agree_{\mu^{(i)}}(u_i,v)\le\agree_{\nu^{(i)}}(\Fold_{c_i}(u_{i-1}),v)$, because
$\mu^{(i)}\le\nu^{(i)}$ and $\mu^{(i)}$ vanishes where $u_i\ne\Fold_{c_i}(u_{i-1})$. Outside
$B_i$ this gives
$\agree_{\mu^{(i)}}(u_i,\Vc_i^{w,D})\le\max\big(\agree_{\mu^{(i-1)}}(u_{i-1},\Vc_{i-1}^{w,D}),\alpha_0\big)$,
and by induction with Step 1, $\agree_{\mu^{(m)}}(u_m,\Vc_m^{w,D})\le\alpha$.

\emph{Step 4 (end).} $\Vc_m^{w,D}$ is the set of constant words with value in
$\F[T]_{\le D}$, and $u_m=g$ is one of them. Hence
$\frac1{|L_m|}\sum_y\mu^{(m)}(y)=\agree_{\mu^{(m)}}(u_m,u_m)\le\alpha$, which by
\cref{ext:claim85} is the probability that one query accepts. The $s$ queries are
independent given the transcript, so outside $\bigcup_iB_i$ all accept with probability
at most $\alpha^s$.
\end{proof}
\uses{\cref{def:protocol,def:weights,lem:round,ext:claim85}}\usedby{the paper}

\begin{remark}[Individual degrees]
Since $C_{w,D}\subseteq\RS[\F,L_0,k^m]$, acceptance also certifies
$\deg_{x_l}<k$ for all $l$; taking $w=e_l$ and $D=d<k$ certifies $\deg_{x_l}f\le d$.
\end{remark}
